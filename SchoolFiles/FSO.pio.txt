; PIO FSO (Free Space Optics) Transceiver
; Uses 4 cycles per bit. Includes Start (1) and Stop (0) bit synchronization.
; TX: Transmits 8 bits (MSB first) 
; RX: Receives 8 bits (MSB first), samples bits precisely in the center.

; Transmitter program
.program fso_tx
    pull block          ; Wait for data to arrive from C (1 cycle)
    set pins, 1 [2]     ; Assert Start Bit high (3 cycles)
    set x, 7            ; Final Start Bit cycle (1 cycle) -> Total 4 cycles
bitloop:
    out pins, 1 [2]     ; Output data bit (3 cycles)
    jmp x--, bitloop    ; Final data bit cycle (1 cycle) -> Total 4 cycles
    set pins, 0 [2]     ; Assert Stop Bit low / idle state (3 cycles)
                        ; Wraps to pull implicitly (1 cycle) -> Total 4 cycles

% c-sdk {
static inline void fso_tx_program_init(PIO pio, uint sm, uint offset, uint pin) {
    // Make sure the laser starts OFF (Low)
    pio_sm_set_pins_with_mask(pio, sm, 0, 1u << pin);
    pio_gpio_init(pio, pin);
    pio_sm_set_consecutive_pindirs(pio, sm, pin, 1, true);
    
    pio_sm_config c = fso_tx_program_get_default_config(offset);
    sm_config_set_out_pins(&c, pin, 1);
    sm_config_set_set_pins(&c, pin, 1);
    // Left-shift (MSB first), Disable autopull (handled manually)
    sm_config_set_out_shift(&c, false, false, 8); 
    
    pio_sm_init(pio, sm, offset, &c);
}
%}

; Receiver program
.program fso_rx
    wait 1 pin 0        ; Wait for Start bit rising edge (Cycle 1)
    nop [1]             ; Start bit Cycles 2, 3
    set x, 7            ; Start bit Cycle 4
bitloop:
    nop [1]             ; Data bit Cycles 1, 2
    in pins, 1          ; Data bit Cycle 3 (Samples perfectly in the middle!)
    jmp x--, bitloop    ; Data bit Cycle 4
    push                ; Send finished byte to C (Stop bit Cycle 1)
    wait 0 pin 0        ; Wait for line to return to low idle (Stop bit Cycle 2)
                        ; Wraps to wait 1 pin 0 (Stop bit Cycles 3, 4)

% c-sdk {
static inline void fso_rx_program_init(PIO pio, uint sm, uint offset, uint pin) {
    pio_gpio_init(pio, pin);
    pio_sm_set_consecutive_pindirs(pio, sm, pin, 1, false);
    
    pio_sm_config c = fso_rx_program_get_default_config(offset);
    sm_config_set_in_pins(&c, pin);
    // Left-shift (MSB first), Disable autopush (handled manually)
    sm_config_set_in_shift(&c, false, false, 8);
    
    pio_sm_init(pio, sm, offset, &c);
}
%}
